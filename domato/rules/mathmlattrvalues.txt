# ============================================================================
# MathML Attribute Values Grammar for Domato / Fuzzing
# ============================================================================
# Based on:
#   - MathML Core (W3C Candidate Recommendation, 2024-2025)
#   - MathML 3.0 2nd Edition (legacy attributes still implemented in browsers)
#   - MDN MathML Reference (https://developer.mozilla.org/en-US/docs/Web/MathML)
#
# This file provides an extended set of value generators for MathML attributes
# to maximize coverage of both modern and legacy browser implementations.
# ============================================================================

# ----------------------------------------------------------------------------
# Basic Types
# ----------------------------------------------------------------------------

# Named references used by MathML token content and identifier-valued attributes.
<identifier> = amp
<identifier> = lt
<identifier> = gt
<identifier> = quot
<identifier> = apos
<identifier> = nbsp
<identifier> = alpha
<identifier> = beta
<identifier> = gamma
<identifier> = infin

# Boolean attributes in MathML (case-insensitive in spec)
<mathmlattr_boolean_value> = true
<mathmlattr_boolean_value> = false

# Unsigned integer (as per MathML Core / CSS Values)
<mathmlattr_unsigned_int_value> = <fuzzint>

<mathmlattr_int_value> = <fuzzint>
<mathmlattr_int_value> = -<fuzzint>
